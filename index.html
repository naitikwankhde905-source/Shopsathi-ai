async function searchProducts() { const message = document.getElementById("input").value.trim(); if (!message) { alert("Apni shopping requirement likho."); return; } const ask = document.getElementById("ask"); const loading = document.getElementById("loading"); const ai = document.getElementById("ai"); const answer = document.getElementById("answer"); const products = document.getElementById("products"); const grid = document.getElementById("grid"); ask.disabled = true; ask.textContent = "🤖 Searching..."; loading.classList.remove("hidden"); ai.classList.add("hidden"); products.classList.add("hidden"); grid.innerHTML = ""; let searchQuery = message; // AI SEARCH try { const aiResponse = await fetch("/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: message }) }); const aiData = await aiResponse.json(); searchQuery = aiData.searchQuery || message; answer.textContent = aiData.reply || "Aapki shopping requirement samajh li."; ai.classList.remove("hidden"); } catch (error) { console.error(error); answer.textContent = "AI abhi available nahi hai. Amazon search ready hai."; ai.classList.remove("hidden"); } // AMAZON SEARCH try { const response = await fetch( "/api/products?q=" + encodeURIComponent(searchQuery) ); const data = await response.json(); if ( data.products && data.products.length > 0 ) { data.products.forEach(product => { const card = document.createElement("div"); card.className = "card"; card.innerHTML = `
🛍️

${escapeHtml(product.title)}

${ product.displayPrice ? `

${escapeHtml( product.displayPrice )}

` : "" } 🛒 View on Amazon `; grid.appendChild(card); }); } else { const url = data.fallbackUrl || ( "https://www.amazon.in/s?k=" + encodeURIComponent(searchQuery) + "&tag=shopsathi01-21" ); grid.innerHTML = `
🛒

Amazon Products

Matching products Amazon par dekhein.

🛒 Search on Amazon
`; } products.classList.remove("hidden"); } catch (error) { console.error(error); grid.innerHTML = `
🛒

Amazon Search

Amazon par products search karein.

🛒 Search on Amazon
`; products.classList.remove("hidden"); } loading.classList.add("hidden"); ask.disabled = false; ask.textContent = "✨ Ask ShopSathi AI"; products.scrollIntoView({ behavior: "smooth" }); }
